# Платформенные реализации добавляются в библиотеку движка: они зависят от целевой ОС.
if(WIN32)
    target_sources(neuexpengine PRIVATE platform_windows.cpp window_validation.cpp resource_path.cpp)
    target_compile_definitions(neuexpengine PRIVATE NOMINMAX)
    target_link_libraries(neuexpengine PUBLIC gdi32 user32 winmm)
elseif(UNIX AND NOT APPLE)
    find_package(X11 REQUIRED)
    target_sources(neuexpengine PRIVATE platform_x11.cpp window_validation.cpp resource_path.cpp)
    target_link_libraries(neuexpengine PUBLIC X11::X11)
else()
    message(FATAL_ERROR "This project currently supports Windows and Linux/X11.")
endif()
